<!-- Litegraph.js (Core) -->
<script src="https://cdn.jsdelivr.net/npm/litegraph.js/build/litegraph.min.js"></script>
<link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/litegraph.js/css/litegraph.css">


<canvas id="mygraph" width="800" height="600"></canvas>

<script>
document.addEventListener("DOMContentLoaded", function () {
    var graph = new LGraph();
    var canvas = new LGraphCanvas("#mygraph", graph);

    // Beispiel: einfacher Knoten
    var node_const = LiteGraph.createNode("basic/const");
    node_const.pos = [200, 200];
    node_const.setValue(4.5);
    graph.add(node_const);

    var node_const2 = LiteGraph.createNode("basic/const");
    node_const2.pos = [400, 200];
    node_const2.setValue(4.5);
    graph.add(node_const2);
	
    graph.start();
});
</script>
