<?php

declare(strict_types=1);

namespace ExtensionBuilder\ExtensionBuilderTypo3\Service;

use Psr\Http\Message\UploadedFileInterface;
use TYPO3\CMS\Core\Http\RequestFactory;
use TYPO3\CMS\Core\Utility\GeneralUtility;
use ZipArchive;

/**
 *
 * Migration:
 * - Target: ExtensionBuilder Core 1.x
 * - Status: legacy
 *
 * @extensionbuilderCoreMajorVersion 0
 * @extensionbuilderMigrationStatus legacy
 *
 * @since 0.14
 */

class AskAiReviewService
//final readonly class AskAiReviewService
{

    /**
     * @since 0.14
     */
    public function __construct(
        private RequestFactory $requestFactory,
    ) {}

    /**
     * @since 0.14
     */
    public function reviewZip(UploadedFileInterface $uploadedFile, string $prompt): string
    {
        $apiKey = getenv('OPENAI_API_KEY');

        if (!$apiKey) {
            throw new \RuntimeException('OPENAI_API_KEY ist nicht gesetzt.', 1740000001);
        }

        $zipPath = $this->storeUploadedZip($uploadedFile);
        $extractPath = GeneralUtility::getFileAbsFileName('var/transient/openai-review-' . bin2hex(random_bytes(8)) . '/');

        GeneralUtility::mkdir_deep($extractPath);

        $this->extractZip($zipPath, $extractPath);

        $codeContext = $this->collectCodeFiles($extractPath);

        $payload = [
            'model' => getenv('OPENAI_MODEL') ?: 'gpt-5.1',
            'input' => [
                [
                    'role' => 'system',
                    'content' => [
                        [
                            'type' => 'input_text',
                            'text' => 'Du bist ein erfahrener TYPO3-Extension-Reviewer. Prüfe Code strukturiert, sachlich und mit konkreten Handlungsempfehlungen.',
                        ],
                    ],
                ],
                [
                    'role' => 'user',
                    'content' => [
                        [
                            'type' => 'input_text',
                            'text' => $prompt . "\n\nHier sind die relevanten Dateien aus der ZIP:\n\n" . $codeContext,
                        ],
                    ],
                ],
            ],
        ];

        $response = $this->requestFactory->request(
            'https://api.openai.com/v1/responses',
            'POST',
            [
                'headers' => [
                    'Authorization' => 'Bearer ' . $apiKey,
                    'Content-Type' => 'application/json',
                ],
                'json' => $payload,
            ],
        );

        $data = json_decode((string)$response->getBody(), true, 512, JSON_THROW_ON_ERROR);

        return $data['output_text'] ?? json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE);
    }

    /**
     * @since 0.14
     */
    private function storeUploadedZip(UploadedFileInterface $uploadedFile): string
    {
        if ($uploadedFile->getError() !== UPLOAD_ERR_OK) {
            throw new \RuntimeException('Upload fehlgeschlagen.', 1740000002);
        }

        $filename = $uploadedFile->getClientFilename() ?? '';

        if (!str_ends_with(strtolower($filename), '.zip')) {
            throw new \RuntimeException('Es sind nur ZIP-Dateien erlaubt.', 1740000003);
        }

        $targetDirectory = GeneralUtility::getFileAbsFileName('var/transient/openai-uploads/');
        GeneralUtility::mkdir_deep($targetDirectory);

        $targetPath = $targetDirectory . bin2hex(random_bytes(8)) . '.zip';

        $uploadedFile->moveTo($targetPath);

        return $targetPath;
    }

    /**
     * @since 0.14
     */
    private function extractZip(string $zipPath, string $extractPath): void
    {
        $zip = new ZipArchive();

        if ($zip->open($zipPath) !== true) {
            throw new \RuntimeException('ZIP-Datei konnte nicht geöffnet werden.', 1740000004);
        }

        for ($i = 0; $i < $zip->numFiles; $i++) {
            $entryName = $zip->getNameIndex($i);

            if ($entryName === false) {
                continue;
            }

            // Schutz gegen Zip Slip
            if (str_contains($entryName, '..') || str_starts_with($entryName, '/')) {
                continue;
            }

            $zip->extractTo($extractPath, [$entryName]);
        }

        $zip->close();
    }

    /**
     * @since 0.14
     */
    private function collectCodeFiles(string $basePath): string
    {
        $allowedExtensions = [
            'php',
            'typoscript',
            'tsconfig',
            'yaml',
            'yml',
            'json',
            'html',
            'htm',
            'js',
            'css',
            'scss',
            'xml',
            'xlf',
            'sql',
            'md',
        ];

        $ignoredPathParts = [
            '/vendor/',
            '/node_modules/',
            '/.git/',
            '/var/',
            '/public/_assets/',
            '/Resources/Public/',
        ];

        $maxBytesPerFile = 80_000;
        $maxTotalBytes = 700_000;

        $result = '';
        $totalBytes = 0;

        $iterator = new \RecursiveIteratorIterator(
            new \RecursiveDirectoryIterator($basePath, \FilesystemIterator::SKIP_DOTS)
        );

        foreach ($iterator as $fileInfo) {
            if (!$fileInfo instanceof \SplFileInfo || !$fileInfo->isFile()) {
                continue;
            }

            $path = $fileInfo->getPathname();
            $normalizedPath = str_replace('\\', '/', $path);

            foreach ($ignoredPathParts as $ignoredPathPart) {
                if (str_contains($normalizedPath, $ignoredPathPart)) {
                    continue 2;
                }
            }

            $extension = strtolower($fileInfo->getExtension());

            if (!in_array($extension, $allowedExtensions, true)) {
                continue;
            }

            if ($fileInfo->getSize() > $maxBytesPerFile) {
                continue;
            }

            $content = file_get_contents($path);

            if ($content === false || !mb_check_encoding($content, 'UTF-8')) {
                continue;
            }

            $relativePath = str_replace($basePath, '', $path);

            $block = "\n\n--- FILE: {$relativePath} ---\n";
            $block .= $content;

            $blockBytes = strlen($block);

            if ($totalBytes + $blockBytes > $maxTotalBytes) {
                $result .= "\n\n--- HINWEIS: Weitere Dateien wurden wegen Größenlimit ausgelassen. ---\n";
                break;
            }

            $result .= $block;
            $totalBytes += $blockBytes;
        }

        return $result;
    }

}
